// ============================================================
// JEST — ESSENCIAL
// ============================================================


// ============================================================
// 1. TEST
// ============================================================
//
// Cria um teste.
//

test("deve somar dois números", () => {
  expect(2 + 2).toBe(4);
});


// ============================================================
// 2. EXPECT
// ============================================================
//
// Define o que esperamos que aconteça.
//

expect(2 + 2).toBe(4);


// Matchers mais importantes:

expect(valor).toBe(10);             // valores simples
expect(objeto).toEqual(objeto);     // objetos/arrays
expect(array).toContain("React");   // contém valor
expect(array).toHaveLength(3);      // tamanho
expect(valor).toBeTruthy();         // verdadeiro
expect(valor).toBeFalsy();          // falso
expect(valor).toBeNull();            // null
expect(valor).toBeUndefined();       // undefined
expect(fn).toThrow();               // lança erro


// ============================================================
// 3. DESCRIBE
// ============================================================
//
// Agrupa testes relacionados.
//

describe("soma", () => {

  test("deve somar", () => {
    expect(2 + 2).toBe(4);
  });

  test("deve funcionar com negativos", () => {
    expect(-2 + -2).toBe(-4);
  });

});


// ============================================================
// 4. BEFORE EACH
// ============================================================
//
// Executa antes de CADA teste.
//

beforeEach(() => {
  // preparação do teste
});


// ============================================================
// 5. MOCK — jest.fn()
// ============================================================
//
// Cria uma função falsa.
//
// Muito importante para testar callbacks.
//

test("deve chamar função", () => {

  const salvar = jest.fn();

  salvar("Vitor");

  expect(salvar).toHaveBeenCalled();
  expect(salvar).toHaveBeenCalledTimes(1);
  expect(salvar).toHaveBeenCalledWith("Vitor");

});


// ============================================================
// 6. MOCK RETURN
// ============================================================
//
// Define o retorno de uma função mock.
//

const buscarNome = jest.fn();

buscarNome.mockReturnValue("Vitor");

expect(buscarNome()).toBe("Vitor");


// ============================================================
// 7. MOCK DE PROMISE
// ============================================================
//
// mockResolvedValue → Promise que funciona
//
// mockRejectedValue → Promise que dá erro
//

const buscarUsuario = jest.fn();

buscarUsuario.mockResolvedValue({
  nome: "Vitor",
});

buscarUsuario.mockRejectedValue(
  new Error("Erro")
);


// ============================================================
// 8. TESTE ASSÍNCRONO
// ============================================================

test("deve buscar usuário", async () => {

  const usuario = await buscarUsuario();

  expect(usuario).toEqual({
    nome: "Vitor",
  });

});


// ============================================================
// 9. JEST.MOCK()
// ============================================================
//
// Substitui um módulo real por um mock.
//
// Muito usado para API, serviços etc.
//

jest.mock("./api");

// ============================================================
// 17. USER EVENT
// ============================================================
//
// Simula ações reais do usuário.
//
// É o que você mais vai usar para interação.
//

await user.type(input, "Vitor");

await user.clear(input);

*/

//
